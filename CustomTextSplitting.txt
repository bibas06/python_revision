from langchain.text_splitter import TextSplitter
from langchain.schema import Document

# Define the CustomTextSplitter class
class CustomTextSplitter(TextSplitter):
    def split_text(self, text):
        # Custom logic for splitting text by paragraphs (double newlines)
        return text.split("\n\n")  

# Create sample documents
documents = [
    Document(
        page_content="This is the first paragraph.\n\nThis is the second paragraph.\n\nThis is the third paragraph.",
        metadata={"source": "doc1"}
    ),
    Document(
        page_content="Another document here.\n\nWith multiple paragraphs.\n\nSplit by blank lines.",
        metadata={"source": "doc2"}
    )
]

# Create splitter instance and split documents
custom_splitter = CustomTextSplitter()
custom_docs = custom_splitter.split_documents(documents)

# Print the results
print("Original documents:")
for i, doc in enumerate(documents):
    print(f"Document {i+1}: {doc.page_content}")
    print(f"Metadata: {doc.metadata}")
    print("---")

print("\nSplit documents:")
for i, doc in enumerate(custom_docs):
    print(f"Chunk {i+1}: {doc.page_content}")
    print(f"Metadata: {doc.metadata}")
    print("---")
